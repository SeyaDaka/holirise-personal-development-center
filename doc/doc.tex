\section{Introduction}

The \textbf{HOLIRISE Personal Development Center} is the official
website of a purpose-driven organization based in Dessie, Ethiopia,
dedicated to personal growth, leadership and community development.
It is built with \textbf{React 18} and \textbf{Vite}.  This document
provides architecture overviews, development guidelines, and
deployment instructions.

\begin{notebox}
This documentation is a living document.  Update it whenever the
codebase changes significantly.
\end{notebox}

\section{Getting Started}

\subsection{Prerequisites}

\begin{itemize}[leftmargin=*, label=\textcolor{accent}{$\bullet$}]
    \item Node.js $\geq$ 18
    \item npm or yarn
\end{itemize}

\subsection{Installation}

\begin{codebox}[Terminal]
\begin{lstlisting}[language=bash]
# Clone the repository
git clone https://github.com/SeyaDaka/holirise-personal-development-center.git
cd holirise-personal-development-center

# Install dependencies
npm install

# Start dev server
npm run dev
\end{lstlisting}
\end{codebox}

\begin{tipbox}
Run \texttt{npm run build} to create an optimised production build
inside \texttt{dist/}.
\end{tipbox}

\section{Content Management}

All site content (hero text, programs, values, team members, contact
details) lives in \texttt{src/data/siteData.js}.  Components read from
this single source of truth, so content changes require no component
edits.

Team member entries support an optional \texttt{photo} field pointing
to an image in \texttt{public/images/}.  When present, the avatar in the Team
section renders the photo; otherwise initials are shown.

\section{Deployment}

\subsection{Deploy to GitHub Pages}

\begin{enumerate}[leftmargin=*, label=\textcolor{accent}{\arabic*.}]
    \item Create a GitHub repository and push your code.
    \item Install the deploy tool:
\end{enumerate}

\begin{codebox}[Terminal]
\begin{lstlisting}[language=bash]
npm install -D gh-pages
\end{lstlisting}
\end{codebox}

\begin{enumerate}[leftmargin=*, label=\textcolor{accent}{\arabic*.}, start=3]
    \item Add deploy scripts to \texttt{package.json}:
\end{enumerate}

\begin{codebox}[package.json]
\begin{lstlisting}[language=JavaScript]
"scripts": {
  "predeploy": "npm run build",
  "deploy": "gh-pages -d dist"
}
\end{lstlisting}
\end{codebox}

\begin{enumerate}[leftmargin=*, label=\textcolor{accent}{\arabic*.}, start=4]
    \item Deploy:
\end{enumerate}

\begin{codebox}[Terminal]
\begin{lstlisting}[language=bash]
npm run deploy
\end{lstlisting}
\end{codebox}

\begin{enumerate}[leftmargin=*, label=\textcolor{accent}{\arabic*.}, start=5]
    \item Go to \textbf{Settings > Pages} in your GitHub repo.
          The site will be live at:
          \texttt{https://<username>.github.io/<repo-name>/}
\end{enumerate}

\begin{notebox}
Repository: \href{https://github.com/SeyaDaka/holirise-personal-development-center}{github.com/SeyaDaka/holirise-personal-development-center}
\end{notebox}

\subsection{Deploy to Vercel}

\begin{enumerate}[leftmargin=*, label=\textcolor{accent}{\arabic*.}]
    \item Push your code to GitHub.
    \item Log in to \href{https://vercel.com}{vercel.com} and click
          \textbf{"Add New Project"}.
    \item Import your GitHub repository.
    \item Vercel auto-detects Vite --- confirm the settings:
\end{enumerate}

\begin{codebox}[Vercel Build Settings]
\begin{lstlisting}
Build command:   npm run build
Output directory: dist
\end{lstlisting}
\end{codebox}

\begin{enumerate}[leftmargin=*, label=\textcolor{accent}{\arabic*.}, start=5]
    \item Click \textbf{Deploy}.  Your site goes live instantly.
    \item Every push to the repository auto-deploys a new version.
\end{enumerate}

\begin{tipbox}[Vercel Tips]
\begin{itemize}[leftmargin=*, label=\textcolor{success}{$\bullet$}]
    \item Custom domains: go to \textbf{Settings > Domains} in the
          Vercel dashboard and add your domain.
    \item Environment variables: add \texttt{.env} values under
          \textbf{Settings > Environment Variables}.
    \item Preview deployments: every pull request gets its own preview URL.
\end{itemize}
\end{tipbox}

\section{Improvements}

\subsection{Performance}
\begin{itemize}[leftmargin=*, label=\textcolor{accent}{$\bullet$}]
    \item \textbf{Image optimization} --- compress team photos using
          \href{https://squoosh.app}{Squoosh} or
          \href{https://tinypng.com}{TinyPNG}.  Use WebP format.
    \item \textbf{Lazy loading} --- add \texttt{loading="lazy"} to
          \texttt{<img>} tags below the fold.
    \item \textbf{Code splitting} --- use \texttt{React.lazy()} and
          \texttt{Suspense} to split components into bundles.
\end{itemize}

\subsection{SEO}
\begin{itemize}[leftmargin=*, label=\textcolor{accent}{$\bullet$}]
    \item Add \texttt{<title>} and \texttt{<meta name="description">}
          in \texttt{index.html}.
    \item Add Open Graph meta tags (\texttt{og:title}, \texttt{og:description},
          \texttt{og:image}) for social sharing.
    \item Create \texttt{robots.txt} and \texttt{sitemap.xml} in
          the \texttt{public/} folder.
\end{itemize}

\subsection{Accessibility}
\begin{itemize}[leftmargin=*, label=\textcolor{accent}{$\bullet$}]
    \item Ensure all images have meaningful \texttt{alt} attributes.
    \item Test keyboard navigation through all interactive elements.
    \item Verify color contrast ratios meet WCAG AA standards.
\end{itemize}

\subsection{Security}
\begin{itemize}[leftmargin=*, label=\textcolor{accent}{$\bullet$}]
    \item Set security headers (\texttt{X-Content-Type-Options},
          \texttt{X-Frame-Options}, \texttt{Content-Security-Policy}).
    \item Enable HTTPS with a free SSL certificate from
          \href{https://letsencrypt.org}{Let's Encrypt}.
\end{itemize}
